cmake_minimum_required(VERSION 3.16)
project(markdownview_host_capability LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(HOST_QSCINTILLA_INCLUDE_DIR "" CACHE PATH "Fixed host QScintilla include directory")
set(HOST_QSCINTILLA_LIBRARY "" CACHE FILEPATH "Fixed host QScintilla static library")
if(NOT EXISTS "${HOST_QSCINTILLA_INCLUDE_DIR}/Qsci/qsciscintilla.h" OR
   NOT EXISTS "${HOST_QSCINTILLA_LIBRARY}")
    message(FATAL_ERROR "Provide the fixed host headers and matching static library")
endif()
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Widgets)
add_library(qt_boundary MODULE qt_boundary.cpp ../../src/source_navigation.cpp)
target_include_directories(qt_boundary PRIVATE ../../src)
target_link_libraries(qt_boundary PRIVATE Qt5::Core Qt5::Gui Qt5::Widgets)
add_executable(host_probe host_probe.cpp)
target_include_directories(host_probe SYSTEM PRIVATE "${HOST_QSCINTILLA_INCLUDE_DIR}")
target_link_libraries(host_probe PRIVATE "${HOST_QSCINTILLA_LIBRARY}" Qt5::Core Qt5::Gui Qt5::Widgets)
foreach(target qt_boundary host_probe)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
